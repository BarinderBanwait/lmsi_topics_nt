\begin{enumerate}

\item Compute the following.
\begin{enumerate}
  \item[(a)] $3^{2027}$ $\Mod{17}$.
  \item[(b)] $7^{1958}$ $\Mod{23}$.
  \item[(c)] $4^{691}$ $\Mod{31}$.
\end{enumerate}

\emph{Hint: Use Fermat's Little theorem.}

\item Find the last digit of $7^{100}$.

\item Prove that $n^7 - n$ is divisible by $42$ for any $n$.

\item Solve the following equations.
\begin{enumerate}
  \item[(a)] $5x \equiv 6 \Mod{8}$.
  \item[(b)] $12x \equiv 5 \Mod{19}$.
  \item[(c)] $x^2 + y^2 \equiv 1 \Mod{13}$.
\end{enumerate}

\item Prove that if $x^3 + y^3 = z^3$ has a solution in integers, then one of the numbers must be a multiple of $7$.

\item \begin{enumerate}
  \item[(a)] Expand $(x+1)^5$ using the binomial theorem.
  \item[(b)] Prove that for any prime $p$, \[ (x+y)^p \equiv x^p + y^p \Mod{p}.\]
\end{enumerate}

\item Modern research into Transformers (the architecture behind LLMs) and SAT Solvers involves translating Logic into Algebra, specifically, polynomials over the field $\F_2$.

In $\F_2$, we only have elements $\{0,1\}$.

\begin{itemize}
  \item False $\to 0$
  \item True $\to 1$
  \item AND($x,y$) $\to x\cdot y$ (multiplication)
  \item NOT(x) $\to 1 + x$.
\end{itemize}

\item \begin{enumerate}
  \item[(a)] Construct a polynomial $P(x,y)$ that represents OR($x,y$).
  \item[(b)] Consider the logic statement: "(x AND y) is True, but (x OR z) is False." Convert this statement into a system of polynomial equations over $\F_2$.
\end{enumerate}

\item Consider the elliptic curve $E$ defined over the finite field $\mathbb{F}_7$ by the equation:
\[
    y^2 \equiv x^3 + 2 \Mod{7}.
\]

\begin{enumerate}
    \item \textbf{Find all affine points:} Determine all pairs $(x, y)$ where $x, y \in \{0, 1, \dots, 6\}$ that satisfy the equation.
    
    \item \textbf{Total count:} Including the ``point at infinity'' (denoted $\mathcal{O}$), how many points are on this curve in total?
\end{enumerate}

\end{enumerate}
