\begin{enumerate}

\item Let $p$ be a prime number.
\begin{enumerate}
  \item[(a)] Find all solutions to $x^2 \equiv 1 \pmod{p}$.
  \item[(b)] For which primes $p$ does $x^2 \equiv -1 \pmod{p}$ have a solution?
  \item[(c)] How many solutions does $x^2 \equiv a \pmod{p}$ have for a given $a \not\equiv 0 \pmod{p}$?
\end{enumerate}

\item Find all solutions to $x^3 + x + 1 \equiv 0 \pmod{5}$.

\item Let $f(x) \in \mathbb{Z}[x]$ be a polynomial of degree $d$. Show that $f(x) \equiv 0 \pmod{p}$ has at most $d$ solutions modulo $p$.

\end{enumerate}
