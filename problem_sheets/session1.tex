\begin{enumerate}

\item Parametrise the rational points on the circle $x^2 + y^2 = 1$.

\item Prove Euclid's theorem: every integral right-angled triangle is of the form
\[
\begin{tikzpicture}[scale=1]
  % Define coordinates (right angle at A)
  \coordinate (A) at (0,0);
  \coordinate (B) at (4,0);
  \coordinate (C) at (0,3);

  % Draw triangle
  \draw (A) -- (B) -- (C) -- cycle;

  % Right angle mark at A
  \draw (A) rectangle ($(A)+(0.3,0.3)$);

  % Label sides
  \node[below] at ($(A)!0.5!(B)$) {$2ab$};
  \node[left] at ($(A)!0.5!(C)$) {$b^2 - a^2$};
  \node[above right] at ($(B)!0.5!(C)$) {$b^2 + a^2$};
\end{tikzpicture}
\]
for $a,b \in \Z$, $a,b > 0$, $0 < a < b$.

\item Prove that the integer $D$ is congruent if and only if the elliptic curve \[ E_D : y^2 = x^3 - D^2x\] admits a rational solution with nonzero $y$-coordinate.

\item Sketch the following curves:
\begin{enumerate}
  \item[(a)] $y^2 = x^3 - D^2x$.
  \item[(b)] $y^2 = x^3 - 3x + 5$.
  \item[(c)] $y^2 = x^3 + 1$.
\end{enumerate}

\item Generate two more points on the elliptic curves in (a) and (c) above.

\item Use Tunnell's theorem to show that $1$ is not a congruent number.

\item Use Tunnell's theorem to write down the first 10 congruent numbers.

\end{enumerate}
