\begin{enumerate}

\item A positive integer $n$ is called a \emph{congruent number} if it is the area of a right triangle with rational side lengths.
\begin{enumerate}
  \item[(a)] Show that $6$ is a congruent number by finding a right triangle with rational sides and area $6$.
  \item[(b)] Show that if $n$ is a congruent number, then so is $k^2 n$ for any positive integer $k$.
  \item[(c)] Is $1$ a congruent number? Justify your answer.
\end{enumerate}

\item Find all Pythagorean triples $(a, b, c)$ with $a, b, c \in \mathbb{Z}_{>0}$ and $\gcd(a, b, c) = 1$ such that $a \leq 20$.

\item Let $E$ be the elliptic curve defined by $y^2 = x^3 - n^2 x$ for a positive integer $n$. Show that if $(x, y)$ is a rational point on $E$ with $y \neq 0$, then $n$ is a congruent number.

\end{enumerate}
